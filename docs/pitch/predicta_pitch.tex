% ---------- Slide 1 · Problema (0:00–0:40) ----------
\begin{frame}{O custo está na forma da curva, não só no volume}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \vspace{2mm}
      \begin{itemize}\setlength\itemsep{2.5mm}
        \item \textbf{Consumo é energia acumulada} (kWh); \textbf{demanda é a potência requerida} em cada instante (kW ou MW).
        \item O mesmo consumo mensal pode formar curvas muito diferentes. O que pressiona a operação é a \textbf{coincidência dos picos}.
        \item Temperatura, umidade, nebulosidade, calendário, hábitos locais e geração distribuída se combinam de modo diferente em cada região.
        \item A tarifa plana não sinaliza ao consumidor quando deslocar a parcela flexível do consumo.
      \end{itemize}
      \vspace{3mm}
      \begin{beamercolorbox}[rounded=true,sep=3mm]{block body}
        \centering\textbf{\textcolor{laranja}{``Não basta prever quanto será consumido.\\ É preciso antecipar quando a carga se concentra.''}}
      \end{beamercolorbox}
    \end{column}
    \begin{column}{0.42\textwidth}
      % Espaço para gráfico: curva de carga com pico destacado (screenshot /territorio ou ilustração)

% ---------- Slide 4 · Confiança (2:25–3:10) ----------
\begin{frame}{Predicta foi mais precisa que o DESSEM}
  \small
  \begin{table}
    \centering
    \begin{tabular}{lrrr}
      \toprule
      \textbf{Subsistema} & \textbf{WAPE DESSEM} & \textbf{WAPE Predicta} & \textbf{menos erro} \\
      \midrule
      Norte & 1,97\% & 1,72\% & \textbf{12,5\%} \\
      Nordeste & 3,46\% & 1,81\% & \textbf{47,7\%} \\
      Sul & 4,07\% & 3,29\% & \textbf{19,2\%} \\
      Sudeste/Centro-Oeste & 3,62\% & 2,30\% & \textbf{36,4\%} \\
      \bottomrule
    \end{tabular}
  \end{table}
  \vspace{2mm}
  \begin{beamercolorbox}[rounded=true,sep=3mm]{block body}
    \centering
    $\displaystyle WAPE=\frac{\sum |\mathrm{realizado}-\mathrm{previsto}|}{\sum |\mathrm{realizado}|}$\qquad
    $\displaystyle ganho=1-\frac{WAPE_{Predicta}}{WAPE_{DESSEM}}$
  \end{beamercolorbox}
  \vspace{2mm}
  \centering
  {\small Mesmas \textbf{672 horas}, mesmo subsistema e mesma carga realizada.}\par
  {\scriptsize\textcolor{cinza}{Comparação histórica com a programação DESSEM publicada. O arquivo foi obtido retrospectivamente; ainda não é um backtest operacional ``as-of''.}}
\end{frame}
